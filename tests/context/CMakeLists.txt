cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(watchdog_common_context)
if(NOT WATCHDOG_CONTEXT_UNIT)
  set(WATCHDOG_CONTEXT_UNIT ${CMAKE_CURRENT_SOURCE_DIR}/../../src/watchdog_context.c)
endif()
target_sources(app PRIVATE main.c ${WATCHDOG_CONTEXT_UNIT} ../../src/watchdog_assert_context.c)
target_include_directories(app PRIVATE ../../include)
target_compile_definitions(app PRIVATE CONFIG_ZMK_WATCHDOG_CAPTURE_CONTEXT=1 CONFIG_ZMK_WATCHDOG_CONTEXT_TEST=1)
zephyr_link_libraries(-Wl,--wrap=z_impl_k_uptime_ticks)
