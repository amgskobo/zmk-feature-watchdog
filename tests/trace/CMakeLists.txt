cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(watchdog_freeze_trace)
target_sources(app PRIVATE main.c ../../src/watchdog_trace.c)
target_sources(app PRIVATE ../../src/watchdog_diagnostics.c ../../src/watchdog_fatal.c ../../src/watchdog_capture.c ../../src/watchdog_context.c)
target_include_directories(app PRIVATE ../../include)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/watchdog_build_id.h "#define WATCHDOG_BUILD_TAG 0x12345678U\n")
target_include_directories(app PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(WATCHDOG_ASSERT_CASE)
  target_compile_definitions(app PRIVATE CONFIG_ZMK_WATCHDOG_ASSERT_CASE=1)
  target_sources(app PRIVATE ../../src/watchdog_assert_context.c)
  zephyr_link_libraries(-Wl,--wrap=assert_post_action)
endif()
